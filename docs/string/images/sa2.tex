\newcommand{\rankrow}[3]{%
  \node at (95,{#1+16}) {rank};
  \foreach \value[count=\i] in {#2} {
    \draw ({138+(\i-1)*70.75},#1) rectangle ({138+\i*70.75},{#1+33});
    \node at ({138+(\i-0.5)*70.75},{#1+16}) {\value};
  }
  \node[chinese,anchor=west] at (735,{#1+16}) {长度为#3};
}
\newcommand{\pairrow}[2]{%
  \node at (80,{#1+16}) {x};
  \node at (120,{#1+16}) {y};
  \foreach \first/\second[count=\i] in {#2} {
    \draw ({138+(\i-1)*70.75},#1) rectangle ({138+\i*70.75},{#1+33});
    \node at ({138+(\i-0.5)*70.75-14},{#1+16}) {\first};
    \node at ({138+(\i-0.5)*70.75+14},{#1+16}) {\second};
  }
}
