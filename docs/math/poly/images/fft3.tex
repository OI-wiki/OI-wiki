\def\pgfsysdriver{pgfsys-dvisvgm.def}
\documentclass[tikz,border=2pt]{standalone}
\usepackage{amsmath}
\usepackage{fontspec}
\usetikzlibrary{arrows.meta}
\newfontfamily\zhfont{Noto Serif CJK SC}
\pagecolor{white}
\begin{document}
\begin{tikzpicture}[>=Stealth,line cap=round,line join=round]
  \def\R{3}
  % 单位圆
  \draw[line width=1.1pt] (0,0) circle[radius=\R];
  % 坐标轴（箭头仅正向）
  \draw[line width=0.9pt,->] (-1.28*\R,0) -- (1.30*\R,0);
  \draw[line width=0.9pt,->] (0,-1.30*\R) -- (0,1.30*\R);
  \fill (0,0) circle[radius=2.2pt];
  % 4 个单位根（红点）
  \foreach \a in {0,90,180,270} \fill[red] (\a:\R) circle[radius=2.6pt];
  % 轴标注
  \node[anchor=south west,align=left,inner sep=1pt] at (0.03,1.24*\R)
    {\Large $y$\\[1pt]{\zhfont 虚数轴}};
  \node[anchor=west,inner sep=1pt] at (1.30*\R,0.10) {\Large $x$};
  \node[anchor=north west,inner sep=1pt] at (1.14*\R,-0.04) {\zhfont 实数轴};
  % 单位根标注
  \node at (0.54*\R,0.20*\R)  {\large $\omega_4^0=1$};
  \node at (0.30*\R,0.60*\R)  {\large $\omega_4^1=i$};
  \node at (-0.52*\R,0.26*\R) {\large $\omega_4^2=-1$};
  \node at (0.34*\R,-0.60*\R) {\large $\omega_4^3=-i$};
\end{tikzpicture}
\end{document}
