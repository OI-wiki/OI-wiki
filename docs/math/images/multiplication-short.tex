% Compile with latex, then dvisvgm --no-fonts --exact --bbox=min.
\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\begin{document}
\definecolor{carryone}{RGB}{235,38,31}
\definecolor{carrytwo}{RGB}{41,175,29}
\definecolor{carrythree}{RGB}{28,164,252}
\definecolor{carryfour}{RGB}{237,98,167}
\definecolor{carryfive}{RGB}{253,147,38}
\scalebox{2}{%
\begin{tikzpicture}[
  x=1.5cm,y=1.2cm,
  digit/.style={circle,draw,very thick,minimum size=10mm,inner sep=1pt,font=\large},
  carry/.style={digit,align=center,font=\normalsize},
  >={Stealth[length=5pt]},very thick
]
  % 1337 * 42 = 56154; each column is a decimal place.
  \foreach \x/\n/\product/\digitcolor in {1/1/42/black,2/3/126/black,3/3/126/black,4/7/294/carryone} {
    \node[digit] (a\x) at (\x,3) {$\n$};
    \node[digit,text=\digitcolor] (b\x) at (\x,1.6) {$\product$};
    \draw[->] (a\x) -- (b\x);
  }
  \node[left] at (0.6,2.3) {$\times 42$};
  % An incoming carry keeps the color of the column to its right.
  % The original operand and plus sign stay black; the total uses this column's color.
  \node[carry] (c0) at (0,0.5) {\scriptsize $0+\textcolor{carryfour}{5}$\\$\textcolor{carryfive}{5}$};
  \node[carry] (c1) at (1,0.5) {\scriptsize $42+\textcolor{carrythree}{14}$\\$\textcolor{carryfour}{56}$};
  \node[carry] (c2) at (2,0.5) {\scriptsize $126+\textcolor{carrytwo}{15}$\\$\textcolor{carrythree}{141}$};
  \node[carry] (c3) at (3,0.5) {\scriptsize $126+\textcolor{carryone}{29}$\\$\textcolor{carrytwo}{155}$};
  \draw[->] (b4) -- (c3);
  \foreach \x/\prev in {3/2,2/1,1/0}
    \draw[->] (c\x) -- (c\prev);
  \foreach \x/\n/\digitcolor in {0/5/carryfive,1/6/carryfour,2/1/carrythree,3/5/carrytwo,4/4/carryone}
    \node[digit,text=\digitcolor] (d\x) at (\x,-0.9) {$\n$};
  \foreach \x in {0,1,2,3}
    \draw (c\x) -- (d\x);
  \draw[->] (b4) -- (d4);
\end{tikzpicture}
}
\end{document}
