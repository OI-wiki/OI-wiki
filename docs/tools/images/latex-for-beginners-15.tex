  \frac{d}{dx} e^x &= e^x \\
  \frac{d}{dx} \int_0^x f(s)\,ds &= f(x) \\
  f(x) &= \sum_{i=0}^{\infty} \frac{f^{(i)}(0)}{i!} x^i \\
  x &= \sqrt{\frac{x_i}{z} y}
